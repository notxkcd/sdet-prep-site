Intermediate

Demonstrate Upcasting

class Parent {
    void show() { System.out.println("Parent's show"); }
}

class Child extends Parent {
    void show() { System.out.println("Child's show"); }
}

public class Main {
    public static void main(String[] args) {
        Parent obj = new Child(); // Upcasting
        obj.show(); // Output: Child's show (Runtime Polymorphism)
    }
}

Demonstrate Abstract class and Private fields with Getters

abstract class Employee {
    private String name; // Encapsulation

    public Employee(String name) { this.name = name; }

    public String getName() { return name; } // Getter

    abstract void work(); // Abstraction
}

class Developer extends Employee {
    public Developer(String name) { super(name); }

    void work() { System.out.println(getName() + " is coding."); }
}

Advanced

Demonstrate the Four Pillars of OOPS

// 1. Abstraction
abstract class Animal {
    // 2. Encapsulation
    private String name;

    public Animal(String name) { this.name = name; }

    public String getName() { return name; }

    public abstract void makeSound();
}

// 3. Inheritance
class Dog extends Animal {
    public Dog(String name) { super(name); }

    // 4. Polymorphism (Overriding)
    @Override
    public void makeSound() {
        System.out.println(getName() + " says: Woof!");
    }
}

public class OOPSDemo {
    public static void main(String[] args) {
        Animal myDog = new Dog("Buddy"); // Upcasting (Polymorphism)
        myDog.makeSound(); 
    }
}
