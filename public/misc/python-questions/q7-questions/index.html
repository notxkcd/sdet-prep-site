Write a program demonstrating Method Overloading (or how to simulate it).

# Python doesn't support traditional overloading, but we can use default args or *args
class Calculator:
    def add(self, a, b, c=0):
        return a + b + c

c = Calculator()
print(c.add(1, 2)) # 3
print(c.add(1, 2, 3)) # 6
