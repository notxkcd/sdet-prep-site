Remove duplicates from a string without using collection concepts.

const str = "hello";
let unique = "";
for (let i = 0; i < str.length; i++) {
    if (unique.indexOf(str[i]) === -1) unique += str[i];
}
console.log(unique); // "helo"

    const client = new MongoClient(url);
