Anagram Check

char[] c1 = str1.toLowerCase().toCharArray();

Count frequency of characters

Map<Character, Integer> freq = new HashMap<>();
for (char c : str.toCharArray()) {
    freq.put(c, freq.getOrDefault(c, 0) + 1);
}
